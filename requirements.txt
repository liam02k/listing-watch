# Plain requests has a Python TLS signature and can get an instant 403.
# Run deploy/diagnose.py to find out which client this machine needs, then set
# EBAY_HTTP_CLIENT accordingly. Install these to give the diagnostic something
# to test -- the monitor auto-selects the best one available.

brotli>=1.1             # so we can honestly advertise Accept-Encoding: br
